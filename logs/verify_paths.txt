  CREATE TABLE `chunk` (
    `chunk_id` bigint(20) NOT NULL,
    `doc_id` varchar(64) NOT NULL,
    `doc_title` varchar(512) DEFAULT NULL,
    `doc_kind` varchar(32) DEFAULT NULL,
    `source_path` varchar(1024) DEFAULT NULL,
    `lang` varchar(8) DEFAULT NULL,
    `page_no` int(11) DEFAULT NULL,
    `section_path` varchar(1024) DEFAULT NULL,
    `line_start` int(11) DEFAULT NULL,
    `line_end` int(11) DEFAULT NULL,
    `chunk_index` int(11) DEFAULT NULL,
    `tags` varchar(512) DEFAULT NULL,
    `content` text DEFAULT NULL,
    `content_hash` char(32) DEFAULT NULL,
    `emb` VECTOR(256) DEFAULT NULL,
    `created_at` datetime DEFAULT NULL,
    KEY `idx_chunk_id` (`chunk_id`) BLOCK_SIZE 16384 LOCAL,
    KEY `idx_doc_id` (`doc_id`) BLOCK_SIZE 16384 LOCAL,
    KEY `idx_doc_kind` (`doc_kind`) BLOCK_SIZE 16384 LOCAL,
    FULLTEXT KEY `ft_chunk` (`content`, `section_path`) WITH PARSER ik PARSER_PROPERTIES=(ik_mode="smart") BLOCK_SIZE 16384,
    VECTOR KEY `v_chunk_emb` (`emb`) WITH (DISTANCE=COSINE, TYPE=HNSW, LIB=VSAG, M=16, EF_CONSTRUCTION=200, EF_SEARCH=64) BLOCK_SIZE 16384,
    FULLTEXT KEY `ft_content` (`content`) WITH PARSER ik PARSER_PROPERTIES=(ik_mode="smart") BLOCK_SIZE 16384,
    FULLTEXT KEY `ft_section` (`section_path`) WITH PARSER ik PARSER_PROPERTIES=(ik_mode="smart") BLOCK_SIZE 16384
  ) ORGANIZATION INDEX DEFAULT CHARSET = utf8mb4 ROW_FORMAT = DYNAMIC COMPRESSION = 'zstd_1.3.8' BLOCK_SIZE = 16384 TABLET_SIZE = 134217728 PCTFREE = 0

    WITH `v` AS (
      SELECT `chunk_id`, (1 - cosine_distance(`emb`, '[0.04847667,0.40445581,-0.00423856,0.01821722,0.02191090,-0.01664348,0.00622679,-0.00142167,0.00029979,-0.02847997,-0.03560242,-0.01130772,0.13647318,0.06936891,0.06542269,-0.06877937,0.05359006,0.01414269,0.02800675,0.00550020,0.03225376,0.06874578,0.06341617,-0.19411674, … 共 256 维])) AS vsim
      FROM `chunk` WHERE 1=1
      ORDER BY cosine_distance(`emb`, '[0.04847667,0.40445581,-0.00423856,0.01821722,0.02191090,-0.01664348,0.00622679,-0.00142167,0.00029979,-0.02847997,-0.03560242,-0.01130772,0.13647318,0.06936891,0.06542269,-0.06877937,0.05359006,0.01414269,0.02800675,0.00550020,0.03225376,0.06874578,0.06341617,-0.19411674, … 共 256 维]) APPROXIMATE LIMIT 60
    ),
    `f` AS (
      SELECT `chunk_id`, MATCH(`content`,`section_path`) AGAINST(%s IN NATURAL LANGUAGE MODE) AS bm25
      FROM `chunk` WHERE MATCH(`content`,`section_path`) AGAINST(%s IN NATURAL LANGUAGE MODE)
      ORDER BY bm25 DESC LIMIT 60
    ),
    `vn` AS (
      SELECT `chunk_id`, vsim, MAX(vsim) OVER () AS vmax,
             ROW_NUMBER() OVER (ORDER BY vsim DESC) AS vrank
      FROM `v`
    ),
    `fn` AS (
      SELECT `chunk_id`, bm25, MAX(bm25) OVER () AS fmax,
             ROW_NUMBER() OVER (ORDER BY bm25 DESC) AS frank
      FROM `f`
    )
    SELECT c.chunk_id, c.doc_id, c.doc_title, c.doc_kind, c.source_path, c.lang,
           c.page_no, c.section_path, c.line_start, c.line_end, c.tags, c.content,
           vn.vsim AS vec_raw, vn.vrank AS vec_rank,
           fn.bm25 AS bm25_raw, fn.frank AS bm25_rank,
           ROUND(COALESCE(vn.vsim / NULLIF(vn.vmax,0), 0), 6) AS vec_score,
           ROUND(COALESCE(fn.bm25 / NULLIF(fn.fmax,0), 0), 6) AS bm25,
           ROUND(0.6 * COALESCE(vn.vsim / NULLIF(vn.vmax,0), 0)
               + 0.4 * COALESCE(fn.bm25 / NULLIF(fn.fmax,0), 0), 6) AS fused
    FROM `vn`
    FULL JOIN `fn` ON vn.chunk_id = fn.chunk_id
    JOIN `chunk` c ON c.chunk_id = COALESCE(vn.chunk_id, fn.chunk_id)
    ORDER BY fused DESC, vec_rank ASC
    LIMIT 5
